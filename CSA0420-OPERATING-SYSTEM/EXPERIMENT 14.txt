#include <stdio.h>

int main()
{
    int n,i;
    char file[20][20];

    printf("Enter Number of Files: ");
    scanf("%d",&n);

    for(i=0;i<n;i++)
    {
        printf("Enter File %d: ",i+1);
        scanf("%s",file[i]);
    }

    printf("\nFiles in Directory:\n");

    for(i=0;i<n;i++)
    {
        printf("%s\n",file[i]);
    }

    return 0;
}
